    """Group blocks with similar y-coordinates into rows."""
    sorted_blocks = sorted(blocks, key=lambda b: b["bbox"]["y0"])

    rows = []
    current_row = []
    current_y = None

    for block in sorted_blocks:
        y = block["bbox"]["y0"]

        if current_y is None or abs(y - current_y) <= tolerance:

            if current_y is None:
                current_y = y
        else:
            rows.append(current_row)
            current_row = [block]
            current_y = y

    return rows

### 3. Column Assignment by X-Coordinate
Mapping text blocks to columns based on horizontal position:

```python
def assign_block_to_column(block, columns):
    """Find which column a block belongs to."""
    block_center = (block["bbox"]["x0"] + block["bbox"]["x1"]) / 2

    best_match = None
    best_distance = float("inf")

    for col_name, col_info in columns.items():
        col_center = col_info["center"]
        distance = abs(block_center - col_center)

        if col_info["x0"] - 50 <= block_center <= col_info["x1"] + 50:
            if distance < best_distance:
                best_distance = distance
                best_match = col_name

    return best_match
```

### 4. Decimal Parsing with Currency Handling
Parsing monetary values with various formats:

```python
from decimal import Decimal, InvalidOperation
import re

def parse_decimal(text: str) -> Optional[Decimal]:
    """Parse decimal from text with currency symbols."""
    # Remove currency symbols
    cleaned = re.sub(r"[$€£¥₱\s]", "", text.strip())

    # Handle comma as thousand/decimal separator
    if "," in cleaned and "." in cleaned:
        # Both present: comma is thousand separator
        cleaned = cleaned.replace(",", "")
    elif "," in cleaned:
        parts = cleaned.split(",")
        if len(parts) == 2 and len(parts[1]) <= 2:
            cleaned = cleaned.replace(",", ".")  # Decimal sep
        else:
            cleaned = cleaned.replace(",", "")  # Thousand sep

    try:
        return Decimal(cleaned)
    except InvalidOperation:
        return None
```

def calculate_confidence(row_data: Dict) -> float:
    """Calculate confidence based on fields found."""
    required_fields = ["description", "quantity", "unit_price", "amount"]
    found_fields = sum(1 for f in required_fields if f in row_data)
    return found_fields / len(required_fields)

def extract_from_document(text_blocks: List[Dict]) -> List[LineItemExtraction]:
    """Full extraction pipeline."""
    # 1. Detect table region
    table_region = detect_table_region(text_blocks)
    if not table_region:
        return []

    # 2. Filter blocks in table region
    table_blocks = [
        b for b in text_blocks

    rows = group_blocks_by_row(table_blocks)
