## Date: 2025-12-18

## Key Learnings

### 1. Setting Up react-pdf
Configuring PDF.js worker for react-pdf:

```typescript
import { Document, Page, pdfjs } from 'react-pdf';

// Set worker from CDN
pdfjs.GlobalWorkerOptions.workerSrc =
  `//unpkg.com/pdfjs-dist@${pdfjs.version}/build/pdf.worker.min.js`;
```

### 2. Document and Page Components
Basic react-pdf structure:

```tsx
<Document
  file={pdfUrl}
  onLoadSuccess={({ numPages }) => setNumPages(numPages)}
  onLoadError={(error) => console.error(error)}

  <Page
    pageNumber={currentPage}
    scale={scale}
    renderTextLayer={false}
    renderAnnotationLayer={false}

```

### 3. Zoom State Management
Managing zoom with min/max limits:

```typescript
const MIN_SCALE = 0.25;
const MAX_SCALE = 3.0;
const SCALE_STEP = 0.25;

const handleZoomIn = () => {
  setScale(prev => Math.min(prev + SCALE_STEP, MAX_SCALE));
};

const handleZoomOut = () => {
  setScale(prev => Math.max(prev - SCALE_STEP, MIN_SCALE));
};

const handleZoomReset = () => {
  setScale(1.0);
};
```

### 4. Page Navigation
Implementing page controls:

```typescript
const handlePrevPage = () => {
  setCurrentPage(prev => Math.max(prev - 1, 1));
};

const handleNextPage = () => {
  setCurrentPage(prev => Math.min(prev + 1, numPages));
};

// Direct page input
const handlePageInputBlur = () => {
  const pageNum = parseInt(pageInputValue, 10);
  if (!isNaN(pageNum) && pageNum >= 1 && pageNum <= numPages) {
    setCurrentPage(pageNum);
  } else {
    setPageInputValue(String(currentPage)); // Reset invalid input
  }
};
```

### 5. Mocking react-pdf in Tests
Creating a mock for react-pdf components:

```typescript
jest.mock('react-pdf', () => ({
  Document: ({ children, onLoadSuccess, file }: any) => {
    React.useEffect(() => {
      if (file) {
        // Simulate async PDF load
        setTimeout(() => {
          onLoadSuccess?.({ numPages: 5 });
        }, 0);
      }
    }, [file, onLoadSuccess]);
    return <div data-testid="pdf-document">{children}</div>;
  },
  Page: ({ pageNumber, scale }: any) => (
    <div data-testid={`pdf-page-${pageNumber}`} data-scale={scale}>
      Page {pageNumber}
    </div>
  ),
  pdfjs: {
    GlobalWorkerOptions: { workerSrc: '' },
  },
}));
```

### 6. Loading State Pattern
Managing loading state with callbacks:

```typescript
const [isLoading, setIsLoading] = useState(true);

const handleLoadSuccess = ({ numPages }: { numPages: number }) => {
  setNumPages(numPages);
  setIsLoading(false);
  onLoadSuccess?.({ numPages });
};

const handleLoadError = (err: Error) => {
  setIsLoading(false);
  onLoadError?.(err);
};
```

### 7. Controlled Input for Page Number
Input that syncs with state:

```typescript
const [pageInputValue, setPageInputValue] = useState('1');

// Update input when page changes
const handleNextPage = () => {
  setCurrentPage(prev => {
    const newPage = Math.min(prev + 1, numPages);
    setPageInputValue(String(newPage));
    return newPage;
  });
};

// Update page when input blurs
const handlePageInputBlur = () => {
  const pageNum = parseInt(pageInputValue, 10);
  if (isValidPage(pageNum)) {
    setCurrentPage(pageNum);
  } else {
    setPageInputValue(String(currentPage)); // Reset to current
  }
};
```

### 10. Disabling Buttons at Boundaries
Disabling navigation at first/last page:

```tsx
<button
  onClick={handlePrevPage}
  disabled={currentPage <= 1 || isLoading}
  className="disabled:opacity-50 disabled:cursor-not-allowed"
>
  Previous
</button>

<button
  onClick={handleNextPage}
  disabled={currentPage >= numPages || isLoading}
  className="disabled:opacity-50 disabled:cursor-not-allowed"
>
  Next
</button>
```
