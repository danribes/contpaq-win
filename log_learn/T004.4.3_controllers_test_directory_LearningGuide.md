# T004.4.3 Learning Guide: ASP.NET Core Controller Testing

## Overview
This guide covers the structure and patterns for testing ASP.NET Core controllers using xUnit, Moq, and FluentAssertions.

## Controller Testing Patterns

### 1. Basic Controller Test Setup
```csharp
public class VendorsControllerTests
{
    private readonly Mock<IVendorService> _mockService;
    private readonly VendorsController _controller;

    public VendorsControllerTests()
    {
        _mockService = new Mock<IVendorService>();
        _controller = new VendorsController(_mockService.Object);
    }
}
```

### 2. Testing GET Endpoints
```csharp
[Fact]
public async Task GetByRfc_WhenExists_ReturnsOkWithVendor()
{
    // Arrange
    var vendor = new VendorDto { Rfc = "ABC123", Name = "Test Vendor" };
    _mockService.Setup(s => s.GetByRfcAsync("ABC123"))
        .ReturnsAsync(vendor);

    // Act
    var result = await _controller.GetByRfc("ABC123");

    // Assert
    var okResult = result.Result.Should().BeOfType<OkObjectResult>().Subject;
    var response = okResult.Value.Should().BeOfType<ApiResponse<VendorDto>>().Subject;
    response.Success.Should().BeTrue();
    response.Data.Rfc.Should().Be("ABC123");
}

[Fact]
public async Task GetByRfc_WhenNotFound_ReturnsNotFound()
{
    // Arrange
    _mockService.Setup(s => s.GetByRfcAsync("NOTEXIST"))
        .ReturnsAsync((VendorDto?)null);

    // Act
    var result = await _controller.GetByRfc("NOTEXIST");

    // Assert
    result.Result.Should().BeOfType<NotFoundObjectResult>();
}
```

### 3. Testing POST Endpoints
```csharp
[Fact]
public async Task Create_WithValidData_ReturnsCreatedAtAction()
{
    // Arrange
    var request = new CreateVendorRequest { Rfc = "ABC123", Name = "New Vendor" };
    var created = new VendorDto { Id = 1, Rfc = "ABC123", Name = "New Vendor" };
    _mockService.Setup(s => s.CreateAsync(It.IsAny<CreateVendorRequest>()))
        .ReturnsAsync(created);

    // Act
    var result = await _controller.Create(request);

    // Assert
    var createdResult = result.Result.Should().BeOfType<CreatedAtActionResult>().Subject;
    createdResult.ActionName.Should().Be(nameof(_controller.GetByRfc));
    var response = createdResult.Value.Should().BeOfType<ApiResponse<VendorDto>>().Subject;
    response.Data.Id.Should().Be(1);
}
```

### 4. Testing Model Validation
```csharp
[Fact]
public async Task Create_WithInvalidModel_ReturnsBadRequest()
{
    // Arrange
    var request = new CreateVendorRequest { Rfc = "", Name = "" };  // Invalid
    _controller.ModelState.AddModelError("Rfc", "RFC is required");

    // Act
    var result = await _controller.Create(request);

    // Assert
    result.Result.Should().BeOfType<BadRequestObjectResult>();
}
```

### 5. Testing Error Scenarios
```csharp
[Fact]
public async Task GetByRfc_WhenServiceThrows_Returns500()
{
    // Arrange
    _mockService.Setup(s => s.GetByRfcAsync(It.IsAny<string>()))
        .ThrowsAsync(new Exception("Database error"));

    // Act
    var result = await _controller.GetByRfc("ABC123");

    // Assert
    var statusResult = result.Result.Should().BeOfType<ObjectResult>().Subject;
    statusResult.StatusCode.Should().Be(500);
}
```

## Best Practices

### 1. Test Each Response Path
```csharp
public class HealthControllerTests
{
    [Fact]
    public async Task GetHealth_WhenAllServicesUp_ReturnsHealthy()
    {
        // Test happy path
    }

    [Fact]
    public async Task GetHealth_WhenDatabaseDown_ReturnsDegraded()
    {
        // Test partial failure
    }

    [Fact]
    public async Task GetHealth_WhenSdkUnavailable_ReturnsUnhealthy()
    {
        // Test complete failure
    }
}
```

### 2. Test Edge Cases
```csharp
[Theory]
[InlineData(null)]
[InlineData("")]
[InlineData("   ")]
public async Task GetByRfc_WithInvalidRfc_ReturnsBadRequest(string? rfc)
{
    var result = await _controller.GetByRfc(rfc!);
    result.Result.Should().BeOfType<BadRequestObjectResult>();
}
```

### 3. Verify Service Interactions
```csharp
[Fact]
public async Task Create_CallsServiceOnce()
{
    // Arrange
    var request = new CreateVendorRequest { Rfc = "ABC123" };
    _mockService.Setup(s => s.CreateAsync(It.IsAny<CreateVendorRequest>()))
        .ReturnsAsync(new VendorDto());

    // Act
    await _controller.Create(request);

    // Assert
    _mockService.Verify(s => s.CreateAsync(It.Is<CreateVendorRequest>(
        r => r.Rfc == "ABC123")), Times.Once);
}
```

## Integration Testing

### WebApplicationFactory Pattern
```csharp
public class VendorsControllerIntegrationTests : IClassFixture<WebApplicationFactory<Program>>
{
    private readonly HttpClient _client;

    public VendorsControllerIntegrationTests(WebApplicationFactory<Program> factory)
    {
        _client = factory.CreateClient();
    }

    [Fact]
    public async Task GetHealth_ReturnsOk()
    {
        var response = await _client.GetAsync("/health");
        response.StatusCode.Should().Be(HttpStatusCode.OK);
    }
}
```

## Test Data Builders
```csharp
public class VendorDtoBuilder
{
    private int _id = 1;
    private string _rfc = "ABC123456789";
    private string _name = "Test Vendor";

    public VendorDtoBuilder WithId(int id) { _id = id; return this; }
    public VendorDtoBuilder WithRfc(string rfc) { _rfc = rfc; return this; }
    public VendorDtoBuilder WithName(string name) { _name = name; return this; }

    public VendorDto Build() => new()
    {
        Id = _id,
        Rfc = _rfc,
        Name = _name
    };
}
```
