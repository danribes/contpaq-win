### 2. PDF Validation
Multiple layers of PDF validation:

```python
def validate_pdf(content: bytes, filename: str) -> None:
    # 1. Check file size
    if len(content) > MAX_FILE_SIZE:
        raise FileTooLargeError("...")

    # 2. Check for empty file
    if len(content) == 0:
        raise InvalidPDFError("El archivo está vacío")

    # 3. Check magic bytes
    if not content.startswith(b"%PDF"):
        raise InvalidPDFError("El archivo no es un PDF válido")

    # 4. Validate structure with PyMuPDF
    import fitz
    doc = fitz.open(stream=content, filetype="pdf")
    if doc.page_count == 0:
        raise InvalidPDFError("El PDF no contiene páginas")
```

### 3. Service Orchestration Pattern
Coordinating multiple services in a pipeline:

```python
class ExtractionService:
    def __init__(self):
        self.pdf_extractor = PDFExtractor()
        self.ocr_service = OCRService()
        self.ai_extractor = AIExtractor()
        self.line_item_extractor = LineItemExtractor()

    def extract(self, content: bytes, filename: str) -> InvoiceExtraction:
        # 1. Detect PDF type
        source_type = self.pdf_extractor.detect_pdf_type(content)

        # 2. Extract text (branch based on type)
        if source_type == SourceType.TEXT:
            text_blocks = self._extract_text_based(content)
        else:
            text_blocks = self._extract_scanned(content)

        # 3. Extract line items
        line_items = self.line_item_extractor.extract_from_document(text_blocks)

        # 4. Run AI extraction
        extraction = self._run_ai_extraction(text_blocks, filename, source_type)

        return extraction
```

### 4. Custom Exception Hierarchy
Defining specific exception types for error handling:

```python
class ExtractionError(Exception):
    """Base exception for extraction errors."""
    pass

class InvalidPDFError(ExtractionError):
    """Raised when the file is not a valid PDF."""
    pass

class FileTooLargeError(ExtractionError):
    """Raised when the file exceeds the maximum size."""
    pass

# Usage in endpoint
try:
    result = service.extract(content, filename)
except InvalidPDFError as e:
    raise HTTPException(status_code=400, detail=str(e))
except FileTooLargeError as e:
    raise HTTPException(status_code=413, detail=str(e))
```
