# Learning Guide: T003.5.4 - Create test utilities and mocks

## Overview

This guide covers creating test utilities, mocks, and fixtures for testing an Electron/React application.

## Concepts Covered

### 1. Mock Organization

Organize mocks by category:

```
tests/mocks/
├── electron.ts     # Electron IPC mocks
├── api.ts          # API service mocks
└── index.ts        # Re-export all mocks
```

### 2. Electron IPC Mocks

Mock the preload API:

```typescript
// tests/mocks/electron.ts
export function createMockElectronAPI() {
  return {
    invoke: jest.fn().mockResolvedValue(undefined),
    send: jest.fn(),
    on: jest.fn(),
    removeListener: jest.fn(),
  };
}

// Setup with default responses
export function setupMockElectronAPI(customResponses = {}) {
  const responses = { ...defaultMockResponses, ...customResponses };
  const mockAPI = createMockElectronAPI();

  mockAPI.invoke.mockImplementation((channel: string) => {
    return Promise.resolve(responses[channel]);
  });

  window.electron = mockAPI;
  return mockAPI;
}
```

### 3. API Service Mocks

Mock each service independently:

```typescript
// tests/mocks/api.ts
export const mockAIService = {
  healthCheck: jest.fn().mockResolvedValue({
    status: 'healthy',
    version: '1.0.0',
  }),

  extractInvoice: jest.fn().mockResolvedValue({
    success: true,
    data: { /* extraction result */ },
  }),

  reset: () => {
    mockAIService.healthCheck.mockClear();
    mockAIService.extractInvoice.mockClear();
  },
};
```

### 4. Custom Render with Providers

Wrap components with required providers:

```tsx
// tests/utils/render.tsx
import { render } from '@testing-library/react';
import { I18nProvider } from '@renderer/i18n';
import { MemoryRouter } from 'react-router-dom';

function AllProviders({ children, route = '/' }) {
  return (
    <I18nProvider>
      <MemoryRouter initialEntries={[route]}>
        {children}
      </MemoryRouter>
    </I18nProvider>
  );
}

export function customRender(ui, options = {}) {
  const { route = '/', ...renderOptions } = options;

  return render(ui, {
    wrapper: ({ children }) => (
      <AllProviders route={route}>{children}</AllProviders>
    ),
    ...renderOptions,
  });
}
```

## Code Examples

### Using Mocks in Tests

```typescript
// Component test with mocks
import { customRender, screen } from '@tests/utils';
import { setupMockElectronAPI } from '@tests/mocks/electron';
import { sampleInvoice } from '@tests/fixtures';
import InvoiceDetails from '@renderer/components/InvoiceDetails';

describe('InvoiceDetails', () => {
  let mockAPI;

  beforeEach(() => {
    mockAPI = setupMockElectronAPI({
      'db:getInvoice': sampleInvoice,
    });
  });

  afterEach(() => {
    jest.clearAllMocks();
  });

  it('displays invoice information', async () => {
    customRender(<InvoiceDetails invoiceId="1" />);

    await screen.findByText('F-001');
    expect(screen.getByText('$11,600.00')).toBeInTheDocument();
  });

  it('calls API to fetch invoice', async () => {
    customRender(<InvoiceDetails invoiceId="1" />);

    expect(mockAPI.invoke).toHaveBeenCalledWith('db:getInvoice', '1');
  });
});
```

### Using Utility Functions

```typescript
import {
  createMockPdfFile,
  createMockDataTransfer,
  flushPromises,
} from '@tests/utils';

describe('FileUpload', () => {
  it('handles file drop', async () => {
    const file = createMockPdfFile('invoice.pdf');
    const dataTransfer = createMockDataTransfer([file]);

    customRender(<FileUpload onUpload={handleUpload} />);

    const dropzone = screen.getByTestId('dropzone');
    fireEvent.drop(dropzone, { dataTransfer });

    await flushPromises();

    expect(handleUpload).toHaveBeenCalledWith(file);
  });
});
```

### Testing with Different Routes

```typescript

describe('Navigation', () => {
  it('renders home page at /', () => {
```

## Best Practices

### 1. Reset Mocks Between Tests

```typescript
beforeEach(() => {
  jest.clearAllMocks();
  // or
  mockAIService.reset();
});
```

### 2. Use Factory Functions for Fixtures

```typescript
// Create variations of sample data
export function createInvoice(overrides: Partial<Invoice> = {}): Invoice {
  return {
    ...sampleInvoice,
    id: generateTestInvoiceId(),
    ...overrides,
  };
}

// Usage
const validatedInvoice = createInvoice({ state: 'VALIDATED' });
```

### 3. Mock at the Right Level

```typescript
// ✅ Good - Mock the API layer
mockAIService.extractInvoice.mockResolvedValue({ ... });

// ❌ Avoid - Mock too deep (implementation details)
jest.mock('../services/aiService', () => ({
  __esModule: true,
  default: { extractInvoice: jest.fn() },
}));
```

### 4. Type Your Mocks

```typescript
interface MockElectronAPI {
  invoke: jest.Mock<Promise<unknown>, [string, ...unknown[]]>;
  send: jest.Mock<void, [string, ...unknown[]]>;
  on: jest.Mock;
  removeListener: jest.Mock;
}

function createMockElectronAPI(): MockElectronAPI {
  return {
    invoke: jest.fn(),
    send: jest.fn(),
    on: jest.fn(),
    removeListener: jest.fn(),
  };
}
```

## Utility Function Reference

| Function | Purpose | Example |
|----------|---------|---------|
| `customRender` | Render with providers | `customRender(<MyComponent />)` |
| `renderWithRoute` | Render at specific route | `renderWithRoute(<App />, '/settings')` |
| `wait` | Delay execution | `await wait(100)` |
| `flushPromises` | Wait for promises | `await flushPromises()` |
| `createMockFile` | Create mock file | `createMockFile('test.pdf', 'application/pdf')` |
| `createMockPdfFile` | Create mock PDF | `createMockPdfFile()` |
| `createMockDataTransfer` | Create drop event data | `createMockDataTransfer([file])` |
| `generateTestRfc` | Generate test RFC | `generateTestRfc('persona_moral')` |

## Summary

Test utilities and mocks provide:
1. ✅ Consistent test setup with custom render
2. ✅ Isolated component testing with mocks
3. ✅ Realistic test data with fixtures
4. ✅ Helper functions for common operations
5. ✅ Type-safe mock implementations
