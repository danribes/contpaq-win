### Existing Logging in handleAutoRestart()

```typescript
private handleAutoRestart(service, code, signal): void {
  // Log when disabled
  if (!this.config.enableAutoRestart) {
    console.log(`[${service}] Auto-restart disabled, not restarting`);
    return;
  }

  // Log when max exceeded
  if (restartCount >= this.config.maxRestarts) {
    console.warn(`[${service}] Max restarts (${this.config.maxRestarts}) exceeded, not restarting`);
    return;
  }

  // Log restart attempt
  console.log(`[${service}] Auto-restarting (attempt ${newRestartCount}/${this.config.maxRestarts}) in ${delay}ms...`);

  // In setTimeout callback, log failures
  setTimeout(() => {
    this.startService().catch((err) => {
      console.error(`[${service}] Auto-restart failed:`, err);
    });
  }, delay);
}
```
